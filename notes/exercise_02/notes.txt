Simple Script to test if everthing is working:
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// Set LCD to 0x27 and 16x2
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Define your pins
const int redBtn    = 2;
const int whiteBtn  = 3;
const int yellowBtn = 4;
const int blackBtn  = 5;
const int buzzer    = 6; // Buzzer moved to Pin 6

void setup() {
  // Initialize LCD
  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("SYSTEM TEST");
  
  // Initialize Buttons (using Internal Pull-up)
  pinMode(redBtn, INPUT_PULLUP);
  pinMode(whiteBtn, INPUT_PULLUP);
  pinMode(yellowBtn, INPUT_PULLUP);
  pinMode(blackBtn, INPUT_PULLUP);
  
  // Initialize Buzzer
  pinMode(buzzer, OUTPUT);
  
  // Quick beep to show it's alive
  digitalWrite(buzzer, HIGH);
  delay(100);
  digitalWrite(buzzer, LOW);
}

void loop() {
  lcd.setCursor(0, 1);
  
  // Read buttons (LOW means pressed because of INPUT_PULLUP)
  bool r = !digitalRead(redBtn);
  bool w = !digitalRead(whiteBtn);
  bool y = !digitalRead(yellowBtn);
  bool b = !digitalRead(blackBtn);

  if (r) {
    lcd.print("RED Button OK   ");
    beep();
  } else if (w) {
    lcd.print("WHITE Button OK ");
    beep();
  } else if (y) {
    lcd.print("YELLOW Button OK");
    beep();
  } else if (b) {
    lcd.print("BLACK Button OK ");
    beep();
  } else {
    lcd.print("Press a button..");
  }
  
  delay(50); // Small delay to stop the LCD from flickering
}

void beep() {
  digitalWrite(buzzer, HIGH);
  delay(20); // Very short beep
  digitalWrite(buzzer, LOW);
}



Red Button: Pin 2 (Menu)

White Button: Pin 3 (Alarm On/Off)

Yellow Button: Pin 4 (Set Hour)

Black Button: Pin 5 (Set Minute)

#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <RTClib.h>
#include <Button.h>

// Musical Notes (Frequencies in Hz)
#define NOTE_C4  262
#define NOTE_G4  392
#define NOTE_A4  440
#define NOTE_B4  494
#define NOTE_C5  523
#define NOTE_E5  659

LiquidCrystal_I2C lcd(0x27, 16, 2); 
RTC_DS1307 rtc;

// PINOUT
const int redBtn    = 2; // Menu Toggle
const int whiteBtn  = 3; // Alarm On/Off
const int yellowBtn = 4; // Set Hour / Preview / STOP
const int blackBtn  = 5; // Set Min / Cycle / SNOOZE
const int buzzerPin = 6; 
const int ledPin    = 13;

// Variables
bool isAlarmSet = false;
bool isRinging = false;
bool didItRingToday = false;
short unsigned int AlarmHH = 12; 
short unsigned int AlarmMM = 0;  
int menuPage = 0; // 0=Clock, 1=Set Time, 2=Set Melody
int selectedMelody = 0; // 0=Classic, 1=Digital, 2=Arpeggio

Button redButton(redBtn);
Button whiteButton(whiteBtn);
Button yellowButton(yellowBtn);
Button blackButton(blackBtn);

void setup() {
  pinMode(buzzerPin, OUTPUT); 
  pinMode(ledPin, OUTPUT);
  lcd.init();
  lcd.backlight();
  rtc.begin();
  
  // rtc.adjust(DateTime(F(__DATE__), F(__TIME__))); // Use once to sync time

  redButton.begin();
  whiteButton.begin();
  yellowButton.begin();
  blackButton.begin();
}

void loop() {
  DateTime now = rtc.now();

  // 1. TRIGGER LOGIC
  if(isAlarmSet && now.hour() == AlarmHH && now.minute() == AlarmMM && !didItRingToday) {
    isRinging = true;
  }

  // 2. ALARM STATE
  if (isRinging) {
    playMelody(selectedMelody);
    if (blackButton.pressed()) snoozeAlarm();
    if (yellowButton.pressed()) stopAlarm();
  } 
  
  // 3. MENU NAVIGATION
  else {
    if (redButton.released()) {
      menuPage++;
      if (menuPage > 2) menuPage = 0;
      lcd.clear();
    }

    if (menuPage == 0) displayClock(now);
    else if (menuPage == 1) setTimePage();
    else if (menuPage == 2) setMelodyPage();
    
    checkAlarmToggle();
  }

  if (now.minute() != AlarmMM) didItRingToday = false;
}

// --- MELODY ENGINE ---
void playMelody(int type) {
  static unsigned long lastNoteTime = 0;
  static int noteIndex = 0;
  
  // Flash Effect
  digitalWrite(ledPin, (millis() % 500 < 250));
  if (millis() % 1000 < 500) lcd.noBacklight(); else lcd.backlight();
  lcd.setCursor(0, 0); lcd.print("!!! WAKE UP !!! ");
  lcd.setCursor(0, 1); lcd.print("YEL:OFF  BLK:SNZ");

  if (millis() - lastNoteTime > 200) {
    lastNoteTime = millis();
    if (type == 0) { // Classic Beep
      tone(buzzerPin, 1000, 100);
    } 
    else if (type == 1) { // Digital Pulse
      int notes[] = {NOTE_E5, 0, NOTE_E5, 0};
      tone(buzzerPin, notes[noteIndex % 4], 100);
    } 
    else { // Arpeggio
      int notes[] = {NOTE_C4, NOTE_E5, NOTE_G4, NOTE_C5};
      tone(buzzerPin, notes[noteIndex % 4], 150);
    }
    noteIndex++;
  }
}

// --- MENU PAGES ---
void displayClock(DateTime t) {
  lcd.setCursor(0, 0);
  lcd.print("Time: ");
  printDigits(t.hour()); lcd.print(":");
  printDigits(t.minute()); lcd.print(":");
  printDigits(t.second());

  lcd.setCursor(0, 1);
  if (isAlarmSet) {
    lcd.print("ALARM "); printDigits(AlarmHH); lcd.print(":"); printDigits(AlarmMM);
    lcd.print(" [M"); lcd.print(selectedMelody); lcd.print("]");
  } else {
    lcd.print("ALARM IS OFF    ");
  }
}

void setTimePage() {
  lcd.setCursor(0, 0); lcd.print("> SET ALARM TIME");
  lcd.setCursor(0, 1);
  printDigits(AlarmHH); lcd.print(":"); printDigits(AlarmMM);
  lcd.print(" (YEL-H BLK-M)");

  if (yellowButton.pressed()) { AlarmHH++; if(AlarmHH>23) AlarmHH=0; }
  if (blackButton.pressed()) { AlarmMM++; if(AlarmMM>59) AlarmMM=0; }
}

void setMelodyPage() {
  lcd.setCursor(0, 0); lcd.print("> SELECT MELODY ");
  lcd.setCursor(0, 1);
  
  if (selectedMelody == 0) lcd.print("0: Classic Beep ");
  if (selectedMelody == 1) lcd.print("1: Digital Pulse");
  if (selectedMelody == 2) lcd.print("2: Arpeggio Fun ");

  if (blackButton.pressed()) {
    selectedMelody++;
    if (selectedMelody > 2) selectedMelody = 0;
    tone(buzzerPin, 400, 50); 
  }

  if (yellowButton.pressed()) {
    lcd.setCursor(0, 1);
    lcd.print("Playing Sample..");
    runPreviewSnippet(selectedMelody);
    lcd.clear(); 
  }
}

// Dedicated function for the "File Preview"
void runPreviewSnippet(int type) {
  for (int i = 0; i < 4; i++) { 
    if (type == 0) tone(buzzerPin, 1000, 200);
    else if (type == 1) tone(buzzerPin, (i % 2 == 0) ? NOTE_E5 : 0, 200);
    else if (type == 2) {
      int sampleNotes[] = {NOTE_C4, NOTE_E5, NOTE_G4, NOTE_C5};
      tone(buzzerPin, sampleNotes[i], 200);
    }
    delay(250); 
  }
  noTone(buzzerPin);
}

// --- UTILS ---
void printDigits(int digits) {
  if (digits < 10) lcd.print('0');
  lcd.print(digits);
}

void checkAlarmToggle() {
  if (whiteButton.released()) {
    isAlarmSet = !isAlarmSet;
    tone(buzzerPin, isAlarmSet ? 800 : 400, 100);
    lcd.clear();
  }
}

void snoozeAlarm() {
  isRinging = false;
  noTone(buzzerPin);
  digitalWrite(ledPin, LOW);
  lcd.backlight();
  AlarmMM += 1; 
  if (AlarmMM >= 60) { AlarmMM = 0; AlarmHH++; }
  lcd.clear(); lcd.print("Snoozing 1 min..");
  delay(2000);
}

void stopAlarm() {
  isRinging = false;
  didItRingToday = true;
  noTone(buzzerPin);
  digitalWrite(ledPin, LOW);
  lcd.backlight();
  lcd.clear(); lcd.print("Alarm Stopped");
  delay(2000);
}

const int echoPin = 9;  
const int trigPin = 10; 

void setup() {
  Serial.begin(9600);
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  pinMode(13, OUTPUT);
  
  // URM37 often likes the Trigger to start HIGH
  digitalWrite(trigPin, HIGH); 
  
  Serial.println("--- URM37 V5 'Deep' Test ---");
}

void loop() {
  // Trigger pulse: LOW for 10 microseconds
  digitalWrite(trigPin, LOW);
  delayMicroseconds(10);
  digitalWrite(trigPin, HIGH);

  // Measure the HIGH pulse on Echo
  // The URM37 can take a while, so we use a 50ms timeout
  long duration = pulseIn(echoPin, LOW, 50000); 
  
  // Note: On some URM37 V5, the Echo signal is INVERTED.
  // If the above doesn't work, we will try: pulseIn(echoPin, HIGH, 50000);

  int distance = duration / 50; // URM37 formula is slightly different

  if (duration == 0) {
    Serial.println("No Pulse Detected. Try swapping Pin 9 and 10?");
  } else {
    Serial.print("Distance: ");
    Serial.print(distance);
    Serial.println(" cm");
    
    // Light up Pin 13 if hand is close
    if (distance > 0 && distance < 20) digitalWrite(13, HIGH);
    else digitalWrite(13, LOW);
  }

  delay(300); 
}

#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <RTClib.h>

#define NOTE_C4  262
#define NOTE_G4  392
#define NOTE_C5  523

LiquidCrystal_I2C lcd(0x27, 16, 2); 
RTC_DS1307 rtc;

const int redBtn     = 2; 
const int whiteBtn   = 3; 
const int yellowBtn  = 4; 
const int blackBtn   = 5; 
const int buzzerPin  = 6; 
const int ledPin     = 13;
const int echoPin    = 9;  
const int trigPin    = 10; 

// State Variables
bool isAlarmSet = false;
bool isRinging = false;
bool didItRingToday = false;
short unsigned int AlarmHH = 12; 
short unsigned int AlarmMM = 0;  
int menuPage = 0; 
int selectedMelody = 0; 

Button redButton(redBtn);
Button whiteButton(whiteBtn);
Button yellowButton(yellowBtn);
Button blackButton(blackBtn);

void setup() {
  pinMode(buzzerPin, OUTPUT); 
  pinMode(ledPin, OUTPUT);
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  
  digitalWrite(trigPin, HIGH); // URM37 default high
  
  lcd.init();
  lcd.backlight();
  rtc.begin();
  
  redButton.begin();
  whiteButton.begin();
  yellowButton.begin();
  blackButton.begin();
}

void loop() {
  DateTime now = rtc.now();

  // 1. ALARM TRIGGER
  if(isAlarmSet && now.hour() == AlarmHH && now.minute() == AlarmMM && !didItRingToday) {
    isRinging = true;
  }

  // 2. ALARM ACTIVE STATE
  if (isRinging) {
    playMelody(selectedMelody);
    
    // GESTURE CONTROL: Check for the wave
    int dist = getDistance();
    if (dist > 0 && dist < 15) { 
      snoozeAlarm(); 
    }
    
    if (blackButton.pressed()) snoozeAlarm();
    if (yellowButton.pressed()) stopAlarm();
  } 
  
  // 3. NORMAL OPERATION / MENU
  else {
    if (redButton.released()) {
      menuPage++;
      if (menuPage > 2) menuPage = 0;
      lcd.clear();
    }

    if (menuPage == 0) displayClock(now);
    else if (menuPage == 1) setTimePage();
    else if (menuPage == 2) setMelodyPage();
    
    checkAlarmToggle();
  }

  if (now.minute() != AlarmMM) didItRingToday = false;
}

// --- URM37 DISTANCE LOGIC ---
int getDistance() {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(10);
  digitalWrite(trigPin, HIGH);
  long duration = pulseIn(echoPin, LOW, 40000); // 40ms timeout
  if (duration == 0) return 999;
  return duration / 50; 
}
